KAYLA'S LIBRARY — EPUB FOCUS MODE FIX
=======================================

PURPOSE
-------

This implementation fixes EPUB pages reverting, repeating, or displaying unrelated
text when entering or leaving Focus Mode.

The central rule is:

There must be exactly ONE EPUB mount and ONE epub.js rendition instance for the
component's lifetime. Focus Mode must never create, replace, unmount, or remount the
EPUB rendition. It only toggles a CSS class on the existing reader container.


1. DELETE THE SECOND EPUB SURFACE
---------------------------------

Remove these completely if they exist:

    epubPreviewMount
    previewRendition
    epub-focus-page
    new EpubRendition(...)

There should be only one call to renderTo() in the reader component.


2. FOCUS MODE ONLY TOGGLES A CLASS

Do not conditionally replace anything inside the EPUB reading surface.

6. PRESERVE THE EXACT CFI WHILE RESIZING
-----------------------------------------

When the toolbar rows change size, the same rendition needs to repaginate. Capture
the exact current CFI before resizing, pass it to resize(), and display that same CFI
after the resize.

File: components/reader.tsx

    useEffect(() => {
      if (book.format !== 'epub' || !epubPageMount.current) return;

      const observedPage = epubPageMount.current;
      let cancelled = false;
      let timer: ReturnType<typeof setTimeout>;

      const schedule = (delay = 160) => {
        if (!epubReady.current) return;

        layoutReflow.current.begin(current.current);
        setSaveState('Fitting page…');
        clearTimeout(timer);

        const resizeAtCurrentPage = async () => {
          const area = epubPageMount.current;
          const r = rendition.current;

          if (cancelled || !area || !r || !epubReady.current) return;

          if (navigationPending.current) {
            timer = setTimeout(() => void resizeAtCurrentPage(), 80);
            return;
          }

          const snapshot = layoutReflow.current.snapshot();
          if (!snapshot) return;

          const { anchor, revision } = snapshot;
          const previous = anchor.location;

          navigationPending.current = true;
          setTurning(true);

          try {
            resizeEpubAt(
              r,
              area.clientWidth,
              area.clientHeight,
              previous || undefined,
            );

            if (previous) await r.display(previous);

            const reported = await reportLatestLocation(r);

            if (
              !cancelled &&
              rendition.current === r &&
              layoutReflow.current.finish(revision)
            ) {
              let sectionCount = 0;
              epub.current?.spine.each(() => sectionCount++);

              requestedCfi.current = previous;
              setAtStart(reported.atStart);
              setAtEnd(reported.atEnd);

              save(
                positionAfterReflow(
                  reported,
                  anchor,
                  chapterData.current,
                  sectionCount,
                  settings.current.fontSize,
                ),
              );

              requestedCfi.current = undefined;
            }
          } catch {
            if (!cancelled && layoutReflow.current.finish(revision)) {
              setSaveState('Page fitting failed — place kept');
              setError(
                'The page could not be fitted to the screen. Your saved place is unchanged.',
              );
            }
          } finally {
            navigationPending.current = false;
            if (!cancelled && rendition.current === r) setTurning(false);
          }
        };

        timer = setTimeout(() => void resizeAtCurrentPage(), delay);
      };

      const observer = new ResizeObserver(() => schedule());
      observer.observe(observedPage);

      return () => {
        cancelled = true;
        observer.disconnect();
        clearTimeout(timer);
      };
    }, [book.format, save]);

8. REPORT THE SETTLED LOCATION
-------------------------------

File: lib/epub-location.ts

    export async function reportLatestLocation(reader: LocationReader) {
      await reader.reportLocation();
      await new Promise<void>((resolve) =>
        requestAnimationFrame(() => resolve()),
      );

      const location = reader.currentLocation() as Location | undefined;

      if (!location?.start?.cfi || !location?.end?.cfi)
        throw new Error('The current page location is not ready.');

      return location;
    }


9. PROTECT THE REFLOW ANCHOR
-----------------------------

File: lib/epub-reflow.ts

    export class EpubReflow {
      private anchor?: Position;
      private revision = 0;

      get pending() {
        return !!this.anchor;
      }

      begin(position: Position) {
        this.anchor ??= structuredClone(position);
        return ++this.revision;
      }

      snapshot() {
        return this.anchor
          ? { anchor: this.anchor, revision: this.revision }
          : undefined;
      }

      finish(revision: number) {
        if (!this.anchor || revision !== this.revision) return false;
        this.anchor = undefined;
        return true;
      }
    }

This prevents rapid ResizeObserver events or toolbar transitions from accepting an
intermediate page as the user's new saved location.

SHORT HANDOFF INSTRUCTION
--------------------------

Delete the epub-focus-page div, epubPreviewMount ref, previewRendition, and every
second EPUB rendition. Keep one permanently mounted epubPageMount and create the
epub.js rendition only once when the book opens. Focus Mode must only toggle a
reader-focus CSS class that hides the existing toolbars. When that changes the
reader's available dimensions, use ResizeObserver to capture the current CFI,
resize the SAME rendition with rendition.resize(width, height, cfi), and restore
the exact passage with rendition.display(cfi). Ignore intermediate relocated events
during this reflow so they cannot overwrite the user's real saved position. Page
turns, bookmarks, and persistence must all read from this single rendition.
